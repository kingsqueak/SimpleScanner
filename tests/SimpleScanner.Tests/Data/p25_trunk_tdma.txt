# Phase 2 calls, for the trunk engine's tests. System 0x123, WACN 0xBEE00, NAC 0x293, RFSS 1, site 1, control channel
# 851.0125 MHz. Identifier 1: 851.00625 MHz + n x 6.25 kHz; 3: TDMA, two slots, the same carriers.
nac 0x293
rfss lra=0x01 flags=1 sysid=0x123 rfss=1 site=1 ch=0x1001 ssc=0x70
net lra=0x01 wacn=0xBEE00 sysid=0x123 ch=0x1001 ssc=0x70
iden id=1 base=851006250 spacing=6250 offset=-45000000 bw=12500
iden_tdma id=3 type=3 base=851006250 spacing=6250 offset=-45000000
grant svc=0x00 ch=0x30A2 tg=1001 src=2001001        # TDMA: 851.5125, slot 0
grant svc=0x00 ch=0x30A3 tg=1002 src=0              # TDMA: 851.5125, slot 1
